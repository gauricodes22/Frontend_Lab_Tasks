let arr = [10, -5, 20, 0, -8, 15, 0, -2]; let positiveSum = 0; let negativeSum = 0; let zeroCount = 0; for (let i = 0; i < arr.length; i++) { if (arr[i] > 0) { positiveSum = positiveSum + arr[i]; } else if (arr[i] < 0) { negativeSum = negativeSum + arr[i]; } else { zeroCount++; } } console.log("Sum of positive numbers:", positiveSum); console.log("Sum of negative numbers:", negativeSum); console.log("Number of zeros:", zeroCount);
